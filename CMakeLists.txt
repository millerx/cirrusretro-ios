cmake_minimum_required(VERSION 3.20)
project(cirrusretro LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
include(FetchContent)

# Keep the audited engine configuration independent of GME's broad defaults.
foreach(format AY GBS GYM HES KSS NSFE SAP)
  set(USE_GME_${format} OFF CACHE BOOL "" FORCE)
endforeach()
set(USE_GME_VGM ON CACHE BOOL "" FORCE)
set(GME_YM2612_EMU GENS CACHE STRING "Audited LGPL-2.1+ implementation" FORCE)
set(USE_GME_SPC ON CACHE BOOL "" FORCE)
set(GME_SPC_ISOLATED_ECHO_BUFFER OFF CACHE BOOL "" FORCE)
set(USE_GME_NSF ON CACHE BOOL "" FORCE)
foreach(option GME_ZLIB GME_BUILD_TESTING GME_BUILD_EXAMPLES GME_BUILD_FRAMEWORK)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
# The Linux regression executable links a private static copy so its
# allocator wrapper cannot affect the CLI or a consumer of the shared GME.
set(test_alloc OFF)
if(BUILD_TESTING AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(test_alloc ON)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(GME_BUILD_STATIC ON CACHE BOOL "" FORCE)
  set(GME_BUILD_SHARED OFF CACHE BOOL "" FORCE)
else()
  set(GME_BUILD_STATIC ${test_alloc} CACHE BOOL "" FORCE)
  set(GME_BUILD_SHARED ON CACHE BOOL "" FORCE)
endif()
FetchContent_Declare(game_music_emu
  GIT_REPOSITORY https://github.com/libgme/game-music-emu.git
  GIT_TAG 9e23d10f9fd2a6a2f33b10912dd8dc7153258995 # 0.6.5
  GIT_SUBMODULES "")
FetchContent_MakeAvailable(game_music_emu)
include(cmake/gme-patches.cmake)

add_library(cirrus_nsf STATIC src/music.c src/vgm.c)
target_include_directories(cirrus_nsf PUBLIC include)
target_link_libraries(cirrus_nsf PRIVATE gme::gme)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  return() # The Xcode app consumes the two static archives.
endif()
add_executable(nsf-to-wav src/nsf_to_wav.c)
target_link_libraries(nsf-to-wav PRIVATE cirrus_nsf)

if(BUILD_TESTING)
  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
  add_test(NAME format-fixtures COMMAND Python3::Interpreter
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/format_fixtures.py ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(format-fixtures PROPERTIES FIXTURES_SETUP music_files)
  add_executable(test-formats tests/formats.c)
  target_link_libraries(test-formats PRIVATE cirrus_nsf)
  add_test(NAME invalid-formats COMMAND Python3::Interpreter
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/invalid_formats.py $<TARGET_FILE:test-formats>)
  set_tests_properties(invalid-formats PROPERTIES TIMEOUT 30)
  add_test(NAME spc COMMAND test-formats ${CMAKE_CURRENT_BINARY_DIR}/pulse.spc spc)
  add_test(NAME vgm COMMAND test-formats ${CMAKE_CURRENT_BINARY_DIR}/pulse.vgm vgm)
  add_test(NAME vgm-dac COMMAND test-formats ${CMAKE_CURRENT_BINARY_DIR}/dac.vgm dac)
  set_tests_properties(spc vgm vgm-dac PROPERTIES FIXTURES_REQUIRED music_files TIMEOUT 10)
  add_executable(test-api tests/api.c)
  target_link_libraries(test-api PRIVATE cirrus_nsf)
  add_test(NAME fixture COMMAND Python3::Interpreter
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture.py ${CMAKE_CURRENT_BINARY_DIR}/pulse.nsf
    ${CMAKE_CURRENT_BINARY_DIR}/warning.nsf)
  set_tests_properties(fixture PROPERTIES FIXTURES_SETUP pulse)
  add_test(NAME c-api COMMAND test-api ${CMAKE_CURRENT_BINARY_DIR}/warning.nsf)
  if(test_alloc)
    add_executable(test-open-alloc tests/open_alloc.c src/music.c src/vgm.c)
    target_include_directories(test-open-alloc PRIVATE include)
    target_link_libraries(test-open-alloc PRIVATE gme_static)
    target_link_options(test-open-alloc PRIVATE "LINKER:--wrap=realloc")
    add_test(NAME open-allocation COMMAND test-open-alloc ${CMAKE_CURRENT_BINARY_DIR}/pulse.nsf)
    foreach(format_file pulse.spc pulse.vgm dac.vgm)
      add_test(NAME allocation-${format_file} COMMAND test-open-alloc ${CMAKE_CURRENT_BINARY_DIR}/${format_file})
      set_tests_properties(allocation-${format_file} PROPERTIES FIXTURES_REQUIRED music_files TIMEOUT 10)
    endforeach()
    set_tests_properties(open-allocation PROPERTIES FIXTURES_REQUIRED pulse TIMEOUT 10)
  endif()
  add_executable(test-native-bounds tests/native_bounds.c ios/App/PCMQueue.c)
  target_include_directories(test-native-bounds PRIVATE ios/App)
  target_link_libraries(test-native-bounds PRIVATE cirrus_nsf)
  add_test(NAME native-bounds COMMAND test-native-bounds ${CMAKE_CURRENT_BINARY_DIR}/pulse.nsf)
  set_tests_properties(native-bounds PROPERTIES FIXTURES_REQUIRED pulse TIMEOUT 5)
  add_test(NAME wav-cli COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/tests/wav_cli.py
    $<TARGET_FILE:nsf-to-wav> ${CMAKE_CURRENT_BINARY_DIR}/pulse.nsf)
  set_tests_properties(c-api wav-cli PROPERTIES FIXTURES_REQUIRED pulse TIMEOUT 60)
endif()
